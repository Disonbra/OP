cmake_minimum_required(VERSION 3.20)
project(OpenMW_iOS_Superbuild)

include(ExternalProject)

# ------------------- Configuration -------------------
option(BUILD_FOR_SIMULATOR "Build for Simulator (ON) or Device (OFF)" ON)

if(BUILD_FOR_SIMULATOR)
    set(IOS_PLATFORM "SIMULATORARM64")
    set(SDK_NAME "iphonesimulator")
else()
    set(IOS_PLATFORM "OS64")
    set(SDK_NAME "iphoneos")
endif()

set(DEPLOYMENT_TARGET "16.2" CACHE STRING "iOS Deployment Target")
set(COMMON_FLAGS "-O3 -fPIC -stdlib=libc++")

# Paths
set(WORK_DIR "${CMAKE_SOURCE_DIR}/ios_build")
set(PREFIX "${WORK_DIR}/ios-libs/${IOS_PLATFORM}")
set(SRC_DIR "${WORK_DIR}/src")
set(PATCHES_DIR "${CMAKE_SOURCE_DIR}/buildscripts/patches")

# Toolchain
set(TOOLCHAIN_DIR "${WORK_DIR}/ios-cmake")
set(TOOLCHAIN_FILE "${TOOLCHAIN_DIR}/ios.toolchain.cmake")

# Ensure toolchain exists
if(NOT EXISTS "${TOOLCHAIN_DIR}")
    execute_process(COMMAND git clone https://github.com/leetal/ios-cmake.git "${TOOLCHAIN_DIR}")
endif()

# ------------------- Helper Macros -------------------

macro(add_dep NAME URL)
    ExternalProject_Add(${NAME}
        URL ${URL}
        PREFIX "${WORK_DIR}/${NAME}"
        STAMP_DIR "${WORK_DIR}/stamps/${NAME}"
        DOWNLOAD_DIR "${SRC_DIR}"
        SOURCE_DIR "${SRC_DIR}/${NAME}"
        CMAKE_ARGS
            -G Xcode
            -DCMAKE_TOOLCHAIN_FILE=${TOOLCHAIN_FILE}
            -DPLATFORM=${IOS_PLATFORM}
            -DDEPLOYMENT_TARGET=${DEPLOYMENT_TARGET}
            -DCMAKE_INSTALL_PREFIX=${PREFIX}
            -DCMAKE_BUILD_TYPE=Release
            -DCMAKE_C_FLAGS=${COMMON_FLAGS}
            -DCMAKE_CXX_FLAGS=${COMMON_FLAGS}
            -DCMAKE_POLICY_VERSION_MINIMUM=3.5
            ${ARGN}
        BUILD_COMMAND cmake --build . --config Release
        INSTALL_COMMAND cmake --install . --config Release
    )
endmacro()

# luajit (Uses custom Makefile)
ExternalProject_Add(luajit
    URL https://github.com/mpvkit/libluajit-build/archive/refs/heads/main.tar.gz
    PREFIX "${WORK_DIR}/luajit"
    SOURCE_DIR "${SRC_DIR}/luajit"
    CONFIGURE_COMMAND ""
    BUILD_COMMAND make build platform=${SDK_NAME} XCFLAGS+="-DLUAJIT_ENABLE_GC64"
    INSTALL_COMMAND
        ${CMAKE_COMMAND} -E copy_directory dist/release/libluajit/include/luajit-2.1/ ${PREFIX}/include/ &&
        ${CMAKE_COMMAND} -E copy dist/release/libluajit/lib/${SDK_NAME}/thin/arm64/lib/libluajit.a ${PREFIX}/lib/
    BUILD_IN_SOURCE 1
)
